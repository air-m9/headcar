    <div id="loading-screen">

        <h2>正在加载 AI 模型...</h2>

    </div>


    <div id="start-screen" class="hidden">

        <h1>头部赛车</h1>

        <p>左右摆头控制赛车移动</p>

        <p>撞击黄色金币得分</p>

        <button onclick="startGame()">开始游戏</button>

    </div>


    <div id="game-over-screen" class="hidden">

        <h1>比赛结束</h1>

        <h2 id="final-score">最终得分: 0</h2>

        <button onclick="resetGame()">再玩一次</button>

    <script>

        // 确保所有函数和变量在全局范围内可用

        const canvas = document.getElementById('game-canvas');

        const ctx = canvas.getContext('2d');

        const videoElement = document.getElementById('video-input');

        

        let width, height;

        let player = { x: 0, y: 0, w: 50, h: 80, color: '#ff4757' };

        let items = [];

        let score = 0;

        let gameTime = 60;

        let isGameRunning = false;

        let headX = 0.5;

        let timerInterval;

        let camera; 


        // 1. 屏幕适配

        function resize() {

            width = window.innerWidth;

            height = window.innerHeight;

            canvas.width = width;

            canvas.height = height;

            player.y = height - 120;

            player.x = width / 2 - player.w / 2;

        }

        window.addEventListener('resize', resize);

        resize();


        // 2. MediaPipe AI 配置

        const faceMesh = new FaceMesh({locateFile: (file) => {

            return `https://cdn.jsdelivr.net/npm/@mediapipe/face_mesh/${file}`;

        }});


        faceMesh.setOptions({

            maxNumFaces: 1,

            refineLandmarks: true,

            minDetectionConfidence: 0.5,

            minTrackingConfidence: 0.5

        });


        faceMesh.onResults(function onResults(results) {

            if (results.multiFaceLandmarks && results.multiFaceLandmarks.length > 0) {

                const landmarks = results.multiFaceLandmarks[0];

                headX = 1 - landmarks[1].x;

            }

        });

        

        // 3. 初始化函数

        function initCameraAndModel() {

            camera = new Camera(videoElement, {

                onFrame: async () => {

                    await faceMesh.send({image: videoElement});

                },

                width: 640,

                height: 480

            });


            camera.start().then(() => {

                document.getElementById('loading-screen').classList.add('hidden');

                document.getElementById('start-screen').classList.remove('hidden');

            }).catch(error => {

                console.error("Camera startup error:", error);

                document.getElementById('loading-screen').innerHTML = `

                    <h2>启动失败！</h2>

                    <p style="color:red;">请确认您是否在 **HTTPS** 链接下访问，并授予摄像头权限。</p>

                `;

            });

        }

        

        // 确保页面加载完成后执行初始化

        window.onload = initCameraAndModel;


        // 4. 游戏逻辑函数 (保持不变)

        function updateScoreUI() {

            document.getElementById('score-display').innerText = `得分: ${score}`;

        }

        

        function startGame() {

            document.getElementById('start-screen').classList.add('hidden');

            document.getElementById('game-over-screen').classList.add('hidden');

            score = 0;

            gameTime = 60;

            items = [];

            isGameRunning = true;

            updateScoreUI();

            

            if (timerInterval) clearInterval(timerInterval);

            timerInterval = setInterval(() => {

                gameTime--;

                document.getElementById('time-display').innerText = `时间: ${gameTime}s`;

                if (gameTime <= 0) gameOver();

            }, 1000);


            requestAnimationFrame(gameLoop);

        }


        function gameOver() {

            isGameRunning = false;

            if (timerInterval) clearInterval(timerInterval);

            document.getElementById('game-over-screen').classList.remove('hidden');

            document.getElementById('final-score').innerText = `最终得分: ${score}`;

        }


        function resetGame() {

            startGame();

        }


        function spawnItem() {

            if (Math.random() < 0.03) { 

                items.push({

                    x: Math.random() * (width - 60) + 30,

                    y: -50,

                    size: 30,

                    collected: false

                });

            }

        }


        function updatePhysics() {

            let targetX = (headX * width) - (player.w / 2);

            player.x += (targetX - player.x) * 0.15; 

            player.x = Math.max(0, Math.min(width - player.w, player.x));


            spawnItem();


            for (let i = 0; i < items.length; i++) {

                let item = items[i];

                item.y += 5; 

                

                // 碰撞检测

                if (!item.collected &&

                    player.x < item.x + item.size/2 &&

                    player.x + player.w > item.x - item.size/2 &&

                    player.y < item.y + item.size/2 &&

                    player.y + player.h > item.y - item.size/2) {

                    

                    item.collected = true;

                    score += 10;

                    updateScoreUI();

                }

            }

            items = items.filter(item => item.y < height && !item.collected);

        }


        function draw() {

            // 绘制道路

            ctx.fillStyle = '#555'; 

            ctx.fillRect(0, 0, width, height);


            // 绘制赛道线

            ctx.strokeStyle = '#fff';

            ctx.lineWidth = 10;

            ctx.setLineDash([40, 40]);

            ctx.beginPath();

            ctx.moveTo(width / 2, 0);

            ctx.lineTo(width / 2, height);

            ctx.stroke();


            // 绘制玩家

            ctx.fillStyle = player.color;

            ctx.fillRect(player.x, player.y, player.w, player.h);

            ctx.fillStyle = '#000';

            ctx.fillRect(player.x + 5, player.y + 15, player.w - 10, 15);


            // 绘制道具

            ctx.fillStyle = 'gold';

            for (let item of items) {

                ctx.beginPath();

                ctx.arc(item.x, item.y, item.size / 2, 0, Math.PI * 2);

                ctx.fill();

            }

        }


        function gameLoop(timestamp) {

            if (!isGameRunning) return;

            updatePhysics();

            draw();

            requestAnimationFrame(gameLoop);

        }


    </script>
